```sql
CASE WHEN e.ScanDoctorName IS NOT NULL AND e.ScanDoctorName <> '' THEN e.ScanDoctorName
     WHEN e.Staff1Name     IS NOT NULL AND e.Staff1Name     <> '' THEN e.Staff1Name
     ELSE e.ExamTechnicianName
END AS EXAM_DOCTOR_NAME,

CASE WHEN e.ScanDoctorName IS NOT NULL AND e.ScanDoctorName <> '' THEN e.ScanDoctorID
     WHEN e.Staff1Name     IS NOT NULL AND e.Staff1Name     <> '' THEN e.Staff1ID
```
